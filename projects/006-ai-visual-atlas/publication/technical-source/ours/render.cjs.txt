const {createRequire}=require('node:module');const fs=require('node:fs');const path=require('node:path');const {spawn}=require('node:child_process');const {pathToFileURL}=require('node:url');
const req=createRequire('[本机配置路径]');const {chromium}=req('playwright');
async function main(){const input=path.resolve(process.argv[2]),output=path.resolve(process.argv[3]);const duration=Number(process.argv[4]||24),fps=24;const frames=Math.ceil(duration*fps);fs.mkdirSync(path.dirname(output),{recursive:true});
const browser=await chromium.launch({executablePath:'[本机配置路径] Files/Google/Chrome/Application/chrome.exe',headless:true});const page=await browser.newPage({viewport:{width:1280,height:720},deviceScaleFactor:1});await page.goto(pathToFileURL(input).href);await page.addStyleTag({content:'#film{width:1280px!important;height:720px!important;border-radius:0!important}main{width:1280px!important;max-width:none!important;padding:0!important}nav{display:none!important}'});await page.evaluate(()=>document.fonts.ready);const errors=[];page.on('pageerror',e=>errors.push(e.message));
const proc=spawn('[本机配置路径]',['-y','-loglevel','error','-f','image2pipe','-vcodec','mjpeg','-framerate',String(fps),'-i','pipe:0','-an','-c:v','libx264','-preset','fast','-crf','19','-pix_fmt','yuv420p','-movflags','+faststart',output],{stdio:['pipe','ignore','pipe']});let stderr='';proc.stderr.on('data',b=>{stderr+=b;console.error(String(b))});proc.stdin.on('error',()=>{});const complete=new Promise((resolve,reject)=>{proc.on('error',reject);proc.on('close',code=>code===0?resolve():reject(new Error(stderr)))});complete.catch(()=>{});const started=Date.now();
for(let i=0;i<frames;i++){await page.evaluate(t=>window.__seek(t),i/fps);const image=await page.locator('#film').screenshot({type:'jpeg',quality:92});if(!proc.stdin.write(image))await new Promise(resolve=>proc.stdin.once('drain',resolve));if(i%120===0)console.log(`frames ${i}/${frames}`)}proc.stdin.end();await complete;await browser.close();if(errors.length)throw new Error(errors.join('\n'));console.log(JSON.stringify({output,frames,fps,duration,elapsedSeconds:(Date.now()-started)/1000}));}
main().catch(e=>{console.error(e);process.exit(1)});
