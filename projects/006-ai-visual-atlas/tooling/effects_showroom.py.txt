def build_credits_html(chrome, provenance):
    rows = []
    for item in provenance['resources']:
        if not item.get('selected_for_publication'):
            continue
        attrs = ''.join('<p>'+escape(text)+'</p>' for text in item['visible_attribution'])
        links = ''.join('<a href="technical-source/effects/'+escape(name,quote=True)+'">'+escape(name)+'</a> ' for name in item['license_documents'])
        rows.append('<section class="research-block" id="'+escape(item['id'],quote=True)+'"><h2>'+escape(item['id'])+'</h2>'+attrs+'<div class="source-ref">'+links+'</div><p><a href="'+escape(item['source_media_url'],quote=True)+'" target="_blank" rel="noreferrer">原作来源</a></p></section>')
    body = '<section class="hero architecture-hero"><div class="eyebrow">ATTRIBUTION / 公开原作的作者与素材署名</div><h1>原作来源与<span>素材署名。</span></h1><p>本次保持原预览内容，完整许可与 CREDITS 随站保留；作者样片与我们的制作分别标示。</p><a class="research-link" href="effects.html">返回完整效果展厅</a></section>'+''.join(rows)
    return chrome('原作者与素材署名 · 效果展厅', body)
