    if filename.startswith('shipvideo-'):

        return ('https://github.com/makevoid/motion-graphics-music-video-skill#videos-created-with-this-skill', '查看作者 MV 来源 ↗')

        return ('https://github.com/JohnHeibel/ClaudeAnimationBase', '查看作者角色动画 ↗')

    return ('production.html#history', '查看本机材料与版本说明 →')


def referenced(href, relative):
    u = urlsplit(html.unescape(href))
    if u.scheme or u.netloc or not u.path:
        return None
    return posixpath.normpath(posixpath.join(str(Path(relative).parent).replace('\\','/'), unquote(u.path)))

    def replace_media(match):
        tag = match.group()
        src = re.search(r'\bsrc=["\']([^"\']+)["\']', tag)
        target = referenced(src.group(1), relative) if src else None
        if target and target not in selected:
            url, label = source_for(target)
            prefix = posixpath.relpath('production.html', str(Path(relative).parent))+'#history' if url.startswith('production.html') else url
            return f'<div class="publication-note"><p>此项为原作来源或未选的本机材料。公开页提供来源与版本说明。</p><a href="{e(prefix)}" target="_blank" rel="noreferrer">{e(label)}</a></div>'
        return re.sub(r'preload=["\'][^"\']+["\']', 'preload="none"', tag)
    text = re.sub(r'<video\b[^>]*>[\s\S]*?</video>|<iframe\b[^>]*>[\s\S]*?</iframe>|<img\b[^>]*>', replace_media, text, flags=re.I)
    def replace_anchor(match):
        anchor = match.group()
        href = re.search(r'\bhref=["\']([^"\']+)["\']', anchor)
        target = referenced(href.group(1), relative) if href else None
        if not target or target in selected:
            return anchor
        url,label=source_for(target)
        if url.startswith('production.html'):
            url=posixpath.relpath('production.html',str(Path(relative).parent))+'#history'
        return f'<a href="{e(url)}" target="_blank" rel="noreferrer">{e(label)}</a>'
    text = re.sub(r'<a\b[^>]*>[\s\S]*?</a>', replace_anchor, text, flags=re.I)
    stylesheet = posixpath.relpath('summary.css', str(Path(relative).parent))
    index = posixpath.relpath('index.html', str(Path(relative).parent))
    text = text.replace('</head>', f'<link rel="stylesheet" href="{stylesheet}"></head>',1)
    note=f'<p class="publication-banner">公开研究版：保留说明和选定自制结果；原作者媒体通过来源入口查阅。本机研究副本、历史中间材料与公开范围分别记录。<a href="{index}#publication-scope">查看范围 →</a></p>'
    return text.replace('<main>', '<main>'+note, 1)


def scrub_json(value):
    if isinstance(value, dict):
        return {k:scrub_json(v) for k,v in value.items() if k not in {'regularFont','boldFont','task_id','taskId'}}
    if isinstance(value,list):return [scrub_json(v) for v in value]
    if isinstance(value,str):
        if re.match(r'^[A-Za-z]:[\\/]',value):return '[本机配置路径已省略]'
    return value
